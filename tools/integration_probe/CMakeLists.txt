cmake_minimum_required(VERSION 3.20)
project(rgbd_integration_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rgbd_camera_interfaces REQUIRED)
find_package(rgbd_object_detector_interfaces REQUIRED)
find_package(rgbd_object_detector REQUIRED)
add_executable(integration_probe probe.cpp)
target_link_libraries(integration_probe PRIVATE rgbd_object_detector::rgbd_detector_overlay)
ament_target_dependencies(integration_probe PUBLIC rclcpp rgbd_camera_interfaces rgbd_object_detector_interfaces)
add_executable(combined_probe combined_probe.cpp)
target_link_libraries(combined_probe PRIVATE rgbd_object_detector::rgbd_detector_overlay)
ament_target_dependencies(combined_probe PUBLIC rclcpp rgbd_camera_interfaces rgbd_object_detector_interfaces)
